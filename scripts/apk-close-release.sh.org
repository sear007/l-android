#!/bin/bash

#Finish release
echo "show env"
env
echo "closing env.."
cd $WORKSPACE
currentVersion=$(cat version)
versionMinor=$(echo ${currentVersion} | rev | cut -d\. -f1 | rev)
versionMajor=${currentVersion: 0: -${#versionMinor} }
newVersion=${versionMajor}$((versionMinor+1))-SNAPSHOT
echo ${newVersion} > version               

 if [ "$GIT_BRANCH" == "alpha" ];then
        nexusRepo=alpha
        echo $nexusRepo
 elif [ "$GIT_BRANCH" == "beta" ];then
         nexusRepo=beta
         echo $nexusRepo
 elif [ "$GIT_BRANCH" == "master" ];then
         nexusRepo=master
         echo $nexusRepo
 fi


git add version
git commit -m "JENKINS: Created new snapshot version"
git push origin HEAD:$nexusRepo
